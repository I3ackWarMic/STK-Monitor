import requests import urllib3 urllib3.disable_warnings(urllib3.exceptions.InsecureRequestWarning) TARGETS = { "pledge1_login": "https://pledge1.bangkok.go.th/sec_Login/", "pawnshop_webboard": "http://pawnshop.bangkok.go.th/webboard/", "pawnshop_index": "http://pawnshop.bangkok.go.th/indexnew.html", } def check_site(name, url, timeout=10): try: resp = requests.get(url, timeout=timeout, verify=False, allow_redirects=True) return { "name": name, "url": url, "status_code": resp.status_code, "up": resp.status_code < 400, "error": None, } except requests.exceptions.RequestException as e: return { "name": name, "url": url, "status_code": None, "up": False, "error": str(e), } def check_all(): results = [] for name, url in TARGETS.items(): results.append(check_site(name, url)) return results if __name__ == "__main__": for r in check_all(): print(r)
